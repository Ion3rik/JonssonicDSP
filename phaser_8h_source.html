60 void prepare(size_t newNumChannels, T newSampleRate) {
61 // Store global parameters
62 numChannels = utils::detail::clampChannels(newNumChannels);
63 sampleRate = utils::detail::clampSampleRate(newSampleRate);
64
65 // Prepare DSP components
66 allPassStage.prepare(numChannels, sampleRate, NUM_STAGES);
67 lfo.prepare(numChannels, sampleRate);
68 lfo.setWaveform(Waveform::Triangle);
69 feedbackSamples.assign(numChannels, T(0));
70
71 // Prepare parameters with smoothing
72 const auto smoothingTime = Time<T>::Milliseconds(T(SMOOTHING_TIME_MS));
73 lfo.setControlSmoothingTime(smoothingTime);
74 modDepthOctaves.prepare(numChannels, sampleRate);
75 centerFrequencyHz.prepare(numChannels, sampleRate);
76 lfoPhaseOffset.prepare(numChannels, sampleRate);
77 feedback.prepare(numChannels, sampleRate);
78 modDepthOctaves.setSmoothingTime(smoothingTime);
79 centerFrequencyHz.setSmoothingTime(smoothingTime);
80 lfoPhaseOffset.setSmoothingTime(smoothingTime);
81 feedback.setSmoothingTime(smoothingTime);
82 modDepthOctaves.setBounds(T(0), MAX_MODULATION_OCTAVES);
83 centerFrequencyHz.setBounds(T(20), sampleRate * T(0.45));
84 lfoPhaseOffset.setBounds(T(0), T(1));
85 feedback.setBounds(-MAX_FEEDBACK, MAX_FEEDBACK);
86
87 // Set default parameters
88 setRate(T(0.5), true);
89 setDepth(T(0.5), true);
90 setFeedback(T(0.25), true);
91 setCenterFrequency(T(1000), true);
92 setSpread(T(0), true);
93 }

109 void processBlock(const T* const* input, T* const* output, size_t numSamples) {
110 for (size_t ch = 0; ch < numChannels; ++ch) {
111 for (size_t n = 0; n < numSamples; ++n) {
112 // Sweep in octaves, smoothing controls before applying the LFO.
113 T lfoValue = lfo.processSample(ch, lfoPhaseOffset.getNextValue(ch));
114 T frequencyHz =
115 centerFrequencyHz.getNextValue(ch) * std::exp2(lfoValue * modDepthOctaves.getNextValue(ch));
116 T drySample = input[ch][n];
117 T feedbackAmount = feedback.getNextValue(ch);
118 T inSample = drySample + feedbackAmount * feedbackSamples[ch];
119 T outSample = allPassStage.processSample(ch, inSample, frequencyHz, T(0.5))[3]; // Allpass output
120 feedbackSamples[ch] = outSample;
121
122 // Blend the input and all-pass output for the classic phaser sound.
123 output[ch][n] = T(0.5) * (drySample + outSample);
124 }
125 }
126 }
