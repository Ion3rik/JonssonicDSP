69 std::array<T, 4> processSample(size_t ch, T input, T f, T Q) {
70 f = std::clamp(f,
71 sampleRate * detail::FilterLimits<T>::MIN_FREQ_NORM,
72 sampleRate * detail::FilterLimits<T>::MAX_FREQ_NORM);
73 Q = detail::FilterLimits<T>::clampQ(Q);
74 T g = std::tan(utils::pi<T> * f / sampleRate);
75 T twoR = T(1) / Q;
76 T d = T(1) / (T(1) + twoR * g + g * g);
77
78 std::array<T, 4> output{};
79 const size_t stateBase = ch * numSections;
80 if constexpr (RoutingType == Routing::Series) {
81 // The first section shares its integrators across all responses.
82 output = processSection(input, g, twoR, d, state[stateBase][0]);
83 // Later sections need independent state for each response cascade.
84 for (size_t section = 1; section < numSections; ++section)
85 for (size_t response = 0; response < output.size(); ++response)
86 output[response] = processSection(output[response], g, twoR, d,
87 state[stateBase + section][response])[response];
88 }
89 if constexpr (RoutingType == Routing::Parallel) {
90 for (size_t section = 0; section < numSections; ++section) {
91 const auto sectionOutput = processSection(input, g, twoR, d, state[stateBase + section][0]);
92 for (size_t response = 0; response < output.size(); ++response)
93 output[response] += sectionOutput[response];
94 }
95 }
96 return output;
97 }

111 void processBlock(const T* const* input,

139 private:
140 // Process one section with the selected cascade's integrator state.
141 static std::array<T, 4> processSection(T input, T g, T twoR, T d, std::array<T, 2>& sectionState) {
142 T hp = (input - sectionState[1] - (twoR + g) * sectionState[0]) * d;
143 T v0 = g * hp;
144 T bp = v0 + sectionState[0];
145 T v1 = g * bp;
146 T lp = v1 + sectionState[1];
147 T ap = hp + lp - twoR * bp;
148 sectionState[0] = bp + v0;
149 sectionState[1] = lp + v1;
150 return {lp, hp, bp, ap};
151 }
152
153 size_t numChannels = 0;
154 size_t numSections = 0;
155 T sampleRate = T(44100);
156 // Two integrator states per response cascade, per channel and section.
157 std::vector<std::array<std::array<T, 2>, 4>> state;
158};
