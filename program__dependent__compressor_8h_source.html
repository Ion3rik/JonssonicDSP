93 T processSample(size_t ch, T input, T detectorInput) {
94 applyProgramDependentControls(ch, gainReductionDb[ch]);
95
96 detectorEnvelope[ch] = envelopeFollower.processSample(ch, detectorInput);
97 gainReductionDb[ch] = gainComputer.processSample(ch, detectorEnvelope[ch]);
98 const T gain = gainSmoother.processSample(ch, gainReductionDb[ch]);
99 maxGainReduction[ch] = std::min(maxGainReduction[ch], gain);
100 return input * gain;
101 }

111 void processBlock(const T* const* input,
112 const T* const* detectorInput,
113 T* const* output,
114 size_t numSamples,
115 T* gainReductionOutput = nullptr) {
116 std::fill(maxGainReduction.begin(), maxGainReduction.end(), T(1));
117 for (size_t ch = 0; ch < numChannels; ++ch) {
118 for (size_t n = 0; n < numSamples; ++n)
119 output[ch][n] = processSample(ch, input[ch][n], detectorInput[ch][n]);
120 if (gainReductionOutput)
121 gainReductionOutput[ch] = maxGainReduction[ch];
122 }
123 }

234 void applyProgramDependentControls(size_t ch, T currentGainReductionDb) {
235 const T reductionDb = std::max(-currentGainReductionDb, T(0));
236 const T amount = std::clamp((reductionDb - modulationMinDb) / (modulationMaxDb - modulationMinDb), T(0), T(1));
237 const auto interpolate = [amount](T minimum, T maximum) { return minimum + amount * (maximum - minimum); };
238
239 const T attackMs = interpolate(attackMinMs, attackMaxMs);
240 if (attackMs != lastAttackMs[ch]) {
241 envelopeFollower.setAttackTime(Time<T>::Milliseconds(attackMs));
242 lastAttackMs[ch] = attackMs;
243 }
244
245 const T releaseMs = interpolate(releaseMinMs, releaseMaxMs);
246 if (releaseMs != lastReleaseMs[ch]) {
247 envelopeFollower.setReleaseTime(Time<T>::Milliseconds(releaseMs));
248 lastReleaseMs[ch] = releaseMs;
249 }
250
251 const T ratio = interpolate(ratioMin, ratioMax);
252 if (ratio != lastRatio[ch]) {
253 gainComputer.setRatio(ratio);
254 lastRatio[ch] = ratio;
255 }
256
257 const T kneeDb = interpolate(kneeMinDb, kneeMaxDb);
258 if (kneeDb != lastKneeDb[ch]) {
259 gainComputer.setKnee(kneeDb);
260 lastKneeDb[ch] = kneeDb;
261 }
262 }
